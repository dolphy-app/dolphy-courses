        function exportStar(a, b) {
          const result = {};
          const names = new Set([...Object.keys(a), ...Object.keys(b)]);
          for (const name of names) {
            if (name === 'default') continue;
            const inA = Object.hasOwn(a, name);
            const inB = Object.hasOwn(b, name);
            if (inA && inB) {
              if (Object.is(a[name], b[name])) result[name] = a[name];
            } else {
              result[name] = inA ? a[name] : b[name];
            }
          }
          return result;
        }

Эмулируйте `export * from a; export * from b`. `exportStar(a, b)` принимает два объекта «экспортов» и возвращает новый объект:

- содержит все собственные имена из `a` и `b`, кроме `default`;
- если имя есть в обоих объектах с **разными** значениями (сравнение `Object.is`), оно неоднозначно и в результат не попадает; с одинаковым значением — попадает;
- `a` и `b` не изменяются.
