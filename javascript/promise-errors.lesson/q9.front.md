            return make();

        function firstSuccess(tasks) {
          if (tasks.length === 0) {
            return Promise.reject(new RangeError('Нет задач'));
          }
          let chain = Promise.reject(new Error('start'));
          for (const task of tasks) {

Напишите функцию `firstSuccess(tasks)`. Задачи (функции, возвращающие промис) запускаются **по очереди**. Результат — значение первой успешно выполненной задачи; последующие не запускаются. Если неудачны все, результат отклоняется ошибкой последней. Пустой массив — отклонение `RangeError`. Исключение, брошенное задачей синхронно, считается неудачей.
